```bash
#!/bin/bash

export SG1684X_TOP=$(cd `dirname ${BASH_SOURCE}`/../ && pwd) # 设置环境变量当前项目路径
export CHIP_ARCH=${1:-bm1684x} # 设置环境变量默认为bm1684
echo "SG1684X_TOP = $SG1684X_TOP" # 打印
echo "CHIP_ARCH = $CHIP_ARCH" 


if [ -z "$TPUKERNEL_FIRMWARE_PATH" ] && [ "$CHIP_ARCH" != "sg2260" ] && [ "$CHIP_ARCH" != "sg2262" ]; then
    export TPUKERNEL_FIRMWARE_PATH=${SG1684X_TOP}/build/firmware_core/libcmodel_firmware.so
fi


if [ -z "$BACKEND_API_PATH" ]; then
    export BACKEND_API_PATH=${SG1684X_TOP}/build/backend_api/libbackend_${CHIP_ARCH}.so
fi
###################################

if [ $CHIP_ARCH = "sg2260" -o "$CHIP_ARCH" = "sg2260e" -o "$CHIP_ARCH" = "sg2262" ]; then
    export TPURT_TOP=$SG1684X_TOP/../tpuv7-runtime
    export PYTHONPATH=${SG1684X_TOP}/tpuDNN/python:${SG1684X_TOP}/tpuDNN/tests_py:${SG1684X_TOP}/tpuDNN/tests_py/test_utils:$PYTHONPATH
    # These envs are here for the tpuv7rt
    export TPU_SCALAER_EMULATOR_WORKDIR=${SG1684X_TOP}/build
    if [ -z "$TPU_EMULATOR_PATH" ]; then
        export TPU_EMULATOR_PATH=${SG1684X_TOP}/build/firmware_core/libcmodel_firmware.so
    fi
fi

# python测试的目录
export TPUDNN_PYTEST_DIR=$SG1684X_TOP/tpuDNN/tests_py
export PYTHONPATH=$TPUDNN_PYTEST_DIR:$PYTHONPATH 

export LIBSOPHON_TOP=$SG1684X_TOP/../libsophon
if [ ! -d "$LIBSOPHON_TOP" ]; then
  export LIBSOPHON_TOP=$SG1684X_TOP/../../libsophon
fi
echo "LIBSOPHON_TOP = $LIBSOPHON_TOP"

if [ ! -d "$LIBSOPHON_TOP" ]; then
  echo "LIBSOPHON_TOP NOT FOUND" 
fi


export PS1="\[\e[1;35m\](${CHIP_ARCH}):\[\e[1;33m\]\w\[\e[1;34m\]\$ \[\e[0m\]"
# 不存在尝试另一个路径
if [ -z "$PREBUILT_DIR" ]; then
	export PREBUILT_DIR=$SG1684X_TOP/../bm_prebuilt_toolchains
fi
# try again
if [ ! -e "$PREBUILT_DIR" ]; then
	export PREBUILT_DIR=$SG1684X_TOP/../../bm_prebuilt_toolchains
fi

if [ ! -e "$PREBUILT_DIR" ]; then
  echo "cannot find 'bm_prebuilt_toolchains' project"
fi

# 设置环境
# 定义这个函数
# 切换环境就直接reload_env会重新加载脚本
function reload_env(){
  if [ -z "$1" ]; then
    source $SG1684X_TOP/scripts/envsetup.sh $CHIP_ARCH
  else
    source $SG1684X_TOP/scripts/envsetup.sh $*
  fi
}

# 设置第三方库路径
THIRD_PARTY_LIBS=
#操作系统的名字，并赋值给变量
OS_TYPE=`cat /etc/os-release | grep ^"NAME=" | cut -d = -f 2`
case ${OS_TYPE[*]:1:-1} in
  "CentOS Linux")
    export THIRD_PARTY_LIBS=$PREBUILT_DIR/x86-64-core-i7--glibc--stable/x86_64-buildroot-linux-gnu/lib64:$THIRD_PARTY_LIBS
    export THIRD_PARTY_LIBS=$PREBUILT_DIR/x86-64-core-i7--glibc--stable/x86_64-buildroot-linux-gnu/sysroot/usr/lib:$THIRD_PARTY_LIBS;;
  *) ;;
esac
export THIRD_PARTY_LIBS=$PREBUILT_DIR/x86-64-core-i7--glibc--stable/lib:$THIRD_PARTY_LIBS

# tvgen的log路径
tv_gen_log_path=$SG1684X_TOP/$CHIP_ARCH/tv_gen/log/
mkdir -p $tv_gen_log_path
export TV_GEN_LOG_PATH=$tv_gen_log_path

# source其他脚本
source ${SG1684X_TOP}/scripts/${CHIP_ARCH}_config.sh
source ${SG1684X_TOP}/scripts/build_helper.sh
source ${SG1684X_TOP}/scripts/build_backend.sh
source ${SG1684X_TOP}/scripts/build_verify.sh
source ${SG1684X_TOP}/scripts/regression.sh
source ${SG1684X_TOP}/scripts/test.sh

unset skip_runtime

# 检查ccache，这个暂时不用管
which ccache &>/dev/null
if [ $? -eq 0 ]; then
    ccache --version
else
    echo
    echo "Ccache not found."
    echo "We highly recommand you to install it via \`sudo apt install ccache\`."
    echo "For the love of mother earth, please."
fi

# 直接就能执行
# 很多的输出打印在终端
# 信息放在日志文件
function quiet_exec()
{
    log=$(echo $@ | md5sum | head -c 20)
    log="/tmp/tpu1686.$log.log"
    $@ 2>&1 | cat > $log
    if [ "${PIPESTATUS[0]}" -ne 0 ]; then
        cat $log
        rm $log
        echo Task failed
        echo $@
    fi
    rm $log
    return ${PIPESTATUS[0]}
}
```
