```c++
template<typename T>
void scatter_data(T* src_ptr, LOCAL_MEM* lmem, TENSOR_INFO* info, u64 lane_mask, bool enable_lane_mask, bool transpose) {
#if defined(SG_TV_GEN) && defined(SG_TV_GEN_DUMP_MASK)
  LOCAL_MEM* p_lmem_mask = get_local_mem_mask(get_cur_nodechip_idx(), ENGINE_BD);
#endif
  ASSERT(info->address % sizeof(T) == 0);
  u32 laddr = (info->address) & ((1 << (LOCAL_MEM_ADDRWIDTH + NPU_SHIFT)) - 1);
  u32 start_npu_idx = laddr >> LOCAL_MEM_ADDRWIDTH;
  u32 start_offset = laddr % LOCAL_MEM_SIZE;
  T* p_src = (T*)src_ptr;

  if (!transpose) {
    for (u32 nidx = 0; nidx < info->n; nidx++) {
      u32 n_offset = nidx * info->n_stride;
      for (u32 cidx = 0; cidx < info->c; cidx++) {
        u32 npu_idx = (start_npu_idx + cidx) % NPU_NUM; 
        if (!enable_lane_mask || (((u64)1 << npu_idx) & lane_mask)) {
          T* npu_mem = (T*)(lmem->mem_arr[npu_idx]); 
          u32 c_offset = ((start_npu_idx + cidx) >> NPU_SHIFT) * info->c_stride; 
          for (u32 hidx = 0; hidx < info->h; hidx++) {
            u32 h_offset = hidx * info->h_stride;
            for (u32 widx = 0; widx < info->w; widx++) {
              if (info->neuron_matrix == 1 && cidx == info->c - 1 &&
                  info->matrix_col_margin != 0 && widx >= info->matrix_col_margin) {
                continue;
              }

              u32 src_offset = widx * info->w_stride +
                              n_offset + c_offset + h_offset;
              src_offset *= sizeof(T);
              src_offset += start_offset;
              ASSERT(src_offset < LOCAL_MEM_SIZE);
              npu_mem[src_offset / sizeof(T)] = *p_src++;
#if defined(SG_TV_GEN) && defined(SG_TV_GEN_DUMP_MASK)
              T* lmem_mask = (T*)p_lmem_mask->mem_arr[npu_idx];
              memset(lmem_mask + src_offset / sizeof(T), 0xff, sizeof(T));
#endif
            } // widx
          } // hidx
        } else {
          p_src += info->h * info->w;
        }
      } // cidx
    } // nidx
  } 
```
