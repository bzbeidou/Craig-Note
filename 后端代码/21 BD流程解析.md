      high = ((u64)FP32 << 3) |
             ((u64)FP32 << 6) |
             ((u64)R_tensor_C << 48);
      WRITE_CMD_BD(0, high, low);
      low = ((u64)R_tensor_W) |
            ((is_L_trans ? (u64)L_col_num : (u64)L_row_num) << 16) |
            ((u64)L_tensor_C << 32) |
            ((u64)L_tensor_W << 48);
      high = ((u64)L_last_W << 16) |
             ((u64)Y_addr << 32);
      WRITE_CMD_BD(1, high, low);
      low = ((u64)L_addr) | ((u64)R_addr << 32);
      high = ((u64)bias_addr);
      WRITE_CMD_BD(2, high, low);
  END_FAST_GEN_CMD(BD, pid_node)

## 1 为什么 bd_cmd_id 占位 1~20 位？不会超出吗？

```
((u64)pid_node->bd_cmd_id << 1)
```

- **位范围**：1~20，总共 20 位。

	**不会超出**，不要多想
