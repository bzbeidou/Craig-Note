### 1. **动态库**

```
// sum.cpp
#include <iostream>

extern "C" __declspec(dllexport) int add(int a, int b) {
    return a + b;
}
```

- `add`  `__declspec(dllexport)` 
- `extern "C"`

#### 编译生成

```
cl /LD sum.cpp  // 使用 /LD 参数生成 DLL
```

参数告诉编译器生成动态库 (`sum.dll`)，而不是静态库或可执行文件。

- **`sum.dll`**
- **`sum.lib`**

```
// main.cpp
#include <iostream>
#include <windows.h>

typedef int (*AddFunc)(int, int);

int main() {
    HINSTANCE hinstLib = LoadLibrary("sum.dll");
    if (hinstLib == NULL) {
        std::cerr << "Failed to load the library!" << std::endl;
        return 1;
    }

    AddFunc add = (AddFunc)GetProcAddress(hinstLib, "add");
    if (add == NULL) {
        std::cerr << "Failed to find the function!" << std::endl;
        return 1;
    }

    std::cout << "Result: " << add(3, 4) << std::endl;

    FreeLibrary(hinstLib);
    return 0;
}
```

```
cl main.cpp  // 编译应用程序
```



### 2. **Linux **

```
// sum.cpp
#include <iostream>

extern "C" int add(int a, int b) {
    return a + b;
}
```

默认情况下，符号是导出的。

```
g++ -fPIC -shared -o libsum.so sum.cpp
```

- `-fPIC`
- `-shared`
- `sum.cpp`

```
// main.cpp
#include <iostream>
#include <dlfcn.h>

typedef int (*AddFunc)(int, int);

int main() {
    void* handle = dlopen("./libsum.so", RTLD_LAZY);
    if (!handle) {
        std::cerr << "Failed to load the library: " << dlerror() << std::endl;
        return 1;
    }

    AddFunc add = (AddFunc)dlsym(handle, "add");
    if (!add) {
        std::cerr << "Failed to find the function: " << dlerror() << std::endl;
        return 1;
    }

    std::cout << "Result: " << add(3, 4) << std::endl;

    dlclose(handle);
    return 0;
}
```

g++ -o main main.cpp -ldl
